app.post("/api/auth/apple", async (req, res) => {
  const clientId  = process.env.APPLE_CLIENT_ID;
  const teamId    = process.env.APPLE_TEAM_ID;
  const keyId     = process.env.APPLE_KEY_ID;
  const privateKey = process.env.APPLE_PRIVATE_KEY?.replace(/\\n/g, "\n");

  if (!clientId || !teamId || !keyId || !privateKey) {
    return res.status(503).json({ error: "Apple sign-in is not configured." });
  }

  const { idToken, firstName, lastName } = req.body;
  if (!idToken) return res.status(400).json({ error: "Missing Apple ID token." });

  const appleSignin = await import("apple-signin-auth");
  const payload = await appleSignin.verifyIdToken(idToken, { audience: clientId });

  const appleId = payload.sub;   // permanent unique Apple user ID
  const email   = payload.email; // may be a relay address like xyz@privaterelay.appleid.com

  let user = await storage.getUserByEmail(email);
  let isNewUser = false;

  if (!user) {
    user = await storage.createUser({
      username: email,
      email,
      password: null as any,

      lastName:  lastName  ?? "User",
      appleId,
      emailVerified: true,
      kycStatus: "pending",
      userTier: "standard",
    });
    isNewUser = true;
  } else if (!(user as any).appleId) {
    await storage.updateUser(user.id, { appleId });
  }

  if (user.accountFrozen) {
    return res.status(403).json({ error: "Account suspended. Contact support." });
  }

  const token = signToken({ userId: user.id, username: user.email, email: user.email });

// POST /api/auth/phone/request
// Accepts phone + email, creates account, sends OTP to email
app.post("/api/auth/phone/request", async (req, res) => {
  try {
    const { phone, email, firstName, lastName } = req.body;

    if (!phone || !/^\+?[1-9]\d{7,14}$/.test(phone.replace(/[\s\-()]/g, ""))) {
      return res.status(400).json({ error: "Please enter a valid phone number." });
    }
    if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
      return res.status(400).json({ error: "Please enter a valid email address." });
    }
    if (!firstName?.trim() || !lastName?.trim()) {
      return res.status(400).json({ error: "Please enter your full name." });
    }

    // Normalise phone — strip spaces/dashes, ensure + prefix
    const normalised = "+" + phone.replace(/[\s\-()]/g, "").replace(/^\+/, "");

    // Check if phone already registered
    const [byPhone] = await db.select().from(users)
      .where(eq(users.phoneNumber, normalised));
    if (byPhone) {
      return res.status(409).json({
        error: "A account with this phone number already exists. Please sign in.",
        existingAccount: true,
      });
    }

    // Check if email already registered
    const existing = await storage.getUserByEmail(email.toLowerCase().trim());
    if (existing) {
      return res.status(409).json({
        error: "An account with this email already exists. Please sign in.",
      });
    }

    // Derive unique username
    const base = email.split("@")[0].toLowerCase().replace(/[^a-z0-9_]/g, "_");
    let username = base;
    let attempt = 0;
    while (await storage.getUserByUsername(username)) {
      attempt++;
      username = `${base}${attempt}`;
    }

    // Create account — no password (phone/email auth)
    const newUser = await storage.createUser({
      username,
      email: email.toLowerCase().trim(),
      password: null as any,
      firstName: firstName.trim(),
      lastName: lastName.trim(),
      phoneNumber: normalised,
      kycStatus: "pending",
      userTier: "standard",
    });

    // Create default AUD wallet
    await storage.createWallet({
      userId: newUser.id,
      currency: "AUD",
      balance: "0.00",
      availableBalance: "0.00",
      walletType: "fiat",
    });

    // Generate OTP — reuse exact same mechanism as email registration
    const verifyToken = randomBytes(32).toString("hex");
    const otp = Math.floor(100000 + Math.random() * 900000).toString();
    const expiry = new Date(Date.now() + 10 * 60 * 1000); // 10 minutes

    await db.update(users).set({
      emailVerificationToken: verifyToken,
      emailVerificationTokenExpiry: expiry,
      emailOtp: otp,
      emailVerified: false,
    }).where(eq(users.id, newUser.id));

    // Send OTP via email (same sendVerificationEmail already used elsewhere)
    const baseUrl = `${req.protocol}://${req.get("host")}`;
    await sendVerificationEmail(
      newUser.email,
      newUser.firstName,
      verifyToken,
      otp,
      baseUrl
    ).catch(() => {});

    await writeAuditLog(
      newUser.id, "phone_register_started", "user",
      String(newUser.id), { phone: normalised, email }, req.ip || null
    );

    const token = signToken({
      userId: newUser.id,
      username: newUser.username,
      email: newUser.email,
    });

    res.status(201).json({
      token,
      userId: newUser.id,
      email: newUser.email,
      emailSent: emailConfigured,
      ...(!emailConfigured ? { devOtp: otp } : {}),
    });
  } catch (err: any) {
    console.error("[phone-request]", err.message);
    if (err.status) return res.status(err.status).json({ error: err.message });
    res.status(500).json({ error: "Registration failed. Please try again." });
  }
});

export const smsConfigured = !!(process.env.VONAGE_API_KEY && process.env.VONAGE_API_SECRET);

export async function sendSmsOtp(
  phone: string,
  otp: string
): Promise<{ sent: boolean }> {
  const apiKey    = process.env.VONAGE_API_KEY;
  const apiSecret = process.env.VONAGE_API_SECRET;

  if (!apiKey || !apiSecret) {
    // Dev fallback — log to console exactly like email OTP does
    console.log(`[sms] OTP for ${phone}: ${otp}`);
    return { sent: false };
  }

  const res = await fetch("https://rest.nexmo.com/sms/json", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      api_key:    apiKey,
      api_secret: apiSecret,
      from:       "AMAX GLOBAL",
      to:         phone.replace(/^\+/, ""),  // Vonage wants no + prefix
      text:       `Your AMAX GLOBAL verification code is: ${otp}. Valid for 10 minutes.`,
    }),
  });

  const data = await res.json() as any;
  const status = data?.messages?.[0]?.status;

  if (status !== "0") {
    console.error("[sms] Vonage error:", data?.messages?.[0]);
    throw new Error("Failed to send SMS. Please check your number and try again.");
  }

  return { sent: true };
}

    const { userId } = requireAuth(req);

    if (!["adviser", "admin", "compliance_admin"].includes(role)) {
      return res.status(403).json({ error: "Adviser access required." });
    }

    // Clients linked to this adviser
    const linkedClients = await db.select({ clientUserId: adviserClients.clientUserId })
      .from(adviserClients)
      .where(eq(adviserClients.adviserUserId, userId));
    const clientIds = linkedClients.map(r => r.clientUserId);

    // KYC pending count — clients who are verified email but not KYC verified
    const kycPending = clientIds.length > 0
      ? await db.select({ id: users.id }).from(users)
          .where(and(
            inArray(users.id, clientIds),
            sql`${users.kycStatus} != 'verified'`
          ))
      : [];

    // Fee consents expiring in 30 days
    const thirtyDaysFromNow = new Date(Date.now() + 30 * 24 * 60 * 60 * 1000);
    const expiringConsents = clientIds.length > 0
      ? await db.select().from(feeConsents)
          .where(and(
            inArray(feeConsents.clientUserId, clientIds),
            eq(feeConsents.status, "active"),
            sql`${feeConsents.expiresAt} <= ${thirtyDaysFromNow}`
          ))
      : [];

    // Open tasks for this adviser
    const openTasks = await db.select().from(adviserTasks)
      .where(and(
        eq(adviserTasks.adviserUserId, userId),
        eq(adviserTasks.status, "open")
      ));

    res.json({
      totalClients:          clientIds.length,
      pendingKyc:            kycPending.length,
      feeConsentsExpiring:   expiringConsents.length,
      openTasks:             openTasks.length,
      openTasksList:         openTasks,
    });
  } catch (err: any) {
    if (err.status) return res.status(err.status).json({ error: err.message });
    res.status(500).json({ error: "Failed to load dashboard." });
  }

// Daily adviser task automation — only triggers backed by real schema data
async function runAdviserTaskAutomation() {
  try {
    const { adviserClients, feeConsents, adviserTasks } = await import("@shared/schema");

    // Helper: create task only if one of this type isn't already open for this client
    async function createTaskIfNotExists(params: {
      adviserUserId: number;
      clientUserId?: number;
      taskType: string;
      title: string;
      priority: string;
      dueAt?: Date;
    }) {
      const where = params.clientUserId
        ? and(
            eq(adviserTasks.adviserUserId, params.adviserUserId),
            eq(adviserTasks.clientUserId, params.clientUserId),
            eq(adviserTasks.taskType, params.taskType),
            eq(adviserTasks.status, "open")
          )
        : and(
            eq(adviserTasks.adviserUserId, params.adviserUserId),
            eq(adviserTasks.taskType, params.taskType),
            eq(adviserTasks.status, "open")
          );
      const [existing] = await db.select({ id: adviserTasks.id }).from(adviserTasks).where(where);
      if (existing) return;
      await db.insert(adviserTasks).values(params);
    }

    // Trigger 1: KYC pending > 24h
    const allLinks = await db.select().from(adviserClients);
    for (const link of allLinks) {
      const [client] = await db.select({
        id: users.id, kycStatus: users.kycStatus, createdAt: users.createdAt,
      }).from(users).where(eq(users.id, link.clientUserId));

      if (!client) continue;
      const ageHours = (Date.now() - new Date(client.createdAt!).getTime()) / (1000 * 60 * 60);
      if (client.kycStatus !== "verified" && ageHours > 24) {
        await createTaskIfNotExists({
          adviserUserId: link.adviserUserId,
          clientUserId:  link.clientUserId,
          taskType:  "kyc_pending",
          title:     "KYC pending — follow up with client",
          priority:  "high",
          dueAt:     new Date(Date.now() + 2 * 24 * 60 * 60 * 1000),
        });
      }

      // Trigger 2: Fee consent expiring within 30 days
      const thirtyDays = new Date(Date.now() + 30 * 24 * 60 * 60 * 1000);
      const expiring = await db.select().from(feeConsents)
        .where(and(
          eq(feeConsents.adviserUserId, link.adviserUserId),
          eq(feeConsents.clientUserId,  link.clientUserId),
          eq(feeConsents.status, "active"),
          sql`${feeConsents.expiresAt} <= ${thirtyDays}`
        ));

      for (const consent of expiring) {
        await createTaskIfNotExists({
          adviserUserId: link.adviserUserId,
          clientUserId:  link.clientUserId,
          taskType:  "fee_consent_expiring",
          title:     `Fee consent expiring ${new Date(consent.expiresAt).toLocaleDateString("en-AU")} — renewal required`,
          priority:  "critical",
          dueAt:     new Date(consent.expiresAt),
        });
      }
    }

    // Trigger 3: Portfolio review every 90 days
    // Uses existing portfolioSnapshots — no phantom tables
    const ninetyDaysAgo = new Date(Date.now() - 90 * 24 * 60 * 60 * 1000);
    for (const link of allLinks) {
      const [recentTask] = await db.select({ id: adviserTasks.id }).from(adviserTasks)
        .where(and(
          eq(adviserTasks.adviserUserId, link.adviserUserId),
          eq(adviserTasks.clientUserId, link.clientUserId),
          eq(adviserTasks.taskType, "portfolio_review"),
          sql`${adviserTasks.createdAt} >= ${ninetyDaysAgo}`
        ));
      if (!recentTask) {
        await createTaskIfNotExists({
          adviserUserId: link.adviserUserId,
          clientUserId:  link.clientUserId,
          taskType:  "portfolio_review",
          title:     "Quarterly portfolio review due",
          priority:  "medium",
          dueAt:     new Date(Date.now() + 7 * 24 * 60 * 60 * 1000),
        });
      }
    }

    console.log("[task-automation] completed at", new Date().toISOString());
  } catch (err: any) {
    console.error("[task-automation] failed:", err.message);
  }
}
