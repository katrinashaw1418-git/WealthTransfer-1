3. Permission rules

Add helper in server/routes.ts:

async function adviserCanAccessClient(adviserUserId: number, clientUserId: number) {
  const [link] = await db
    .select()
    .from(adviserClients)
    .where(and(
      eq(adviserClients.adviserUserId, adviserUserId),
      eq(adviserClients.clientUserId, clientUserId)
    ));

  return !!link;
}

Every adviser client route must check this.
