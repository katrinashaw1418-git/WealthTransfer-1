type RiskAnswers = {
  marketDropReaction: "sell_all" | "sell_some" | "hold" | "buy_more";
  volatilityTolerance: "low" | "some" | "moderate" | "high";
  lossTolerance: "under_5" | "5_10" | "10_20" | "over_20";
  investmentExperience: "none" | "basic" | "moderate" | "advanced";
  incomeReliance: "full" | "partial" | "none";

  incomeStability: "stable" | "variable" | "unstable";
  liquidityBufferMonths: number;
  dependantsCount: number;
  debtRatio: "low" | "medium" | "high";
  investmentHorizon: "<2" | "2-5" | "5-10" | "10+";
  liquidityNeeds: "low" | "medium" | "high";
};

export function scoreRiskProfile(input: RiskAnswers) {
  let behaviouralScore = 0;

  behaviouralScore += {
    sell_all: 0,
    sell_some: 2,
    hold: 4,
    buy_more: 6,
  }[input.marketDropReaction];

  behaviouralScore += {
    low: 1,
    some: 3,
    moderate: 5,
    high: 7,
  }[input.volatilityTolerance];

  behaviouralScore += {
    under_5: 1,
    "5_10": 3,
    "10_20": 5,
    over_20: 7,
  }[input.lossTolerance];

  behaviouralScore += {
    none: 1,
    basic: 3,
    moderate: 5,
    advanced: 7,
  }[input.investmentExperience];

  behaviouralScore += {
    full: 1,
    partial: 3,
    none: 5,
  }[input.incomeReliance];

  let capacityAdjustment = 0;

  capacityAdjustment += {
    stable: 2,
    variable: 0,
    unstable: -2,
  }[input.incomeStability];

  if (input.liquidityBufferMonths > 12) capacityAdjustment += 2;
  else if (input.liquidityBufferMonths >= 6) capacityAdjustment += 1;
  else capacityAdjustment -= 2;

  if (input.dependantsCount === 0) capacityAdjustment += 1;
  else if (input.dependantsCount >= 3) capacityAdjustment -= 2;

  capacityAdjustment += {
    low: 1,
    medium: 0,
    high: -2,
  }[input.debtRatio];

  let finalScore = behaviouralScore + capacityAdjustment;
  finalScore = Math.max(5, Math.min(35, finalScore));

  let riskBand = scoreToRiskBand(finalScore);

  const overrideReasons: string[] = [];

  if (input.investmentHorizon === "<2" && riskRank(riskBand) > riskRank("moderate")) {
    riskBand = "moderate";
    overrideReasons.push("Investment horizon is less than 2 years; maximum risk profile capped at Moderate.");
  }

  if (input.liquidityNeeds === "high" && riskRank(riskBand) > riskRank("balanced")) {
    riskBand = "balanced";
    overrideReasons.push("High liquidity need; maximum risk profile capped at Balanced.");
  }

  if (input.incomeReliance === "full" && riskRank(riskBand) > riskRank("balanced")) {
    riskBand = "balanced";
    overrideReasons.push("Client relies on portfolio income; maximum risk profile capped at Balanced.");
  }

  const recommendedPortfolio = riskBand;
  const allocation = mapPortfolioAllocation(recommendedPortfolio);

  return {
    behaviouralScore,
    capacityAdjustment,
    finalScore,
    riskBand,
    recommendedPortfolio,
    overrideApplied: overrideReasons.length > 0,
    overrideReasons,
    allocation,
    scoringInputs: input,
  };
}

function scoreToRiskBand(score: number) {
  if (score <= 10) return "conservative";
  if (score <= 16) return "moderate";
  if (score <= 22) return "balanced";
  if (score <= 28) return "growth";
  return "high_growth";
}

function riskRank(band: string) {
  return {
    conservative: 1,
    moderate: 2,
    balanced: 3,
    growth: 4,
    high_growth: 5,
  }[band] ?? 1;
}

function mapPortfolioAllocation(portfolio: string) {
  const allocations = {
    conservative: {
      cash: 25,
      bonds: 45,
      equities: 25,
      alternatives: 5,
      crypto: 0,
    },
    moderate: {
      cash: 15,
      bonds: 35,
      equities: 45,
      alternatives: 5,
      crypto: 0,
    },
    balanced: {
      cash: 10,
      bonds: 25,
      equities: 55,
      alternatives: 5,
      crypto: 5,
    },
    growth: {
      cash: 5,
      bonds: 10,
      equities: 70,
      alternatives: 5,
      crypto: 10,
    },
    high_growth: {
      cash: 0,
      bonds: 5,
      equities: 75,
      alternatives: 5,
      crypto: 15,
    },
  };

  return allocations[portfolio as keyof typeof allocations] ?? allocations.balanced;
}
