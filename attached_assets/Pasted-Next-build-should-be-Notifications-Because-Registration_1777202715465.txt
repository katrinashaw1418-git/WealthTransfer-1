app.get("/api/adviser/notifications", async (req, res) => {
  const { userId } = requireAuth(req);
  const role = await getRole(userId);

  if (role !== "adviser") {
    return res.status(403).json({ error: "Adviser access required" });
  }

  const notifications = await db
    .select()
    .from(adviserNotifications)
    .where(eq(adviserNotifications.adviserUserId, userId))
    .orderBy(desc(adviserNotifications.createdAt));

  res.json({ notifications });
});
GET count
app.get("/api/adviser/notifications/count", async (req, res) => {
  const { userId } = requireAuth(req);
  const role = await getRole(userId);

  if (role !== "adviser") {
    return res.status(403).json({ error: "Adviser access required" });
  }

  const unread = await db
    .select()
    .from(adviserNotifications)
    .where(and(
      eq(adviserNotifications.adviserUserId, userId),
      eq(adviserNotifications.status, "unread")
    ));

  res.json({ unreadCount: unread.length });
});
Mark read
app.patch("/api/adviser/notifications/:id/read", async (req, res) => {
  const { userId } = requireAuth(req);
  const id = Number(req.params.id);

  const [updated] = await db
    .update(adviserNotifications)
    .set({
      status: "read",
      readAt: new Date(),
    })
    .where(and(
      eq(adviserNotifications.id, id),
      eq(adviserNotifications.adviserUserId, userId)
    ))
    .returning();

  if (!updated) {
    return res.status(404).json({ error: "Notification not found" });
  }

  res.json({ notification: updated });
});

3. Notification generation logic

Create service:

server/services/adviser-notifications.ts
export async function createNotificationIfMissing({
  adviserUserId,
  clientUserId,
  type,
  title,
  message,
  priority,
  deepLink,
  relatedEntityType,
  relatedEntityId,
}: InsertAdviserNotification) {
  const [existing] = await db
    .select()
    .from(adviserNotifications)
    .where(and(
      eq(adviserNotifications.adviserUserId, adviserUserId),
      eq(adviserNotifications.type, type),
      eq(adviserNotifications.relatedEntityType, relatedEntityType),
      eq(adviserNotifications.relatedEntityId, relatedEntityId),
      eq(adviserNotifications.status, "unread")
    ));

  if (existing) return existing;

  const [created] = await db
    .insert(adviserNotifications)
    .values({
      adviserUserId,
      clientUserId,
      type,
      title,
      message,
      priority,
      deepLink,
      relatedEntityType,
      relatedEntityId,
    })
    .returning();

  return created;
}

Component

Create:

client/src/components/adviser/notification-bell.tsx
import { Bell } from "lucide-react";
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import { useLocation } from "wouter";
import { Button } from "@/components/ui/button";

export default function AdviserNotificationBell() {
  const [, navigate] = useLocation();
  const queryClient = useQueryClient();

  const { data } = useQuery({
    queryKey: ["/api/adviser/notifications"],
    queryFn: async () => {
      const res = await fetch("/api/adviser/notifications");
      if (!res.ok) throw new Error("Failed to load notifications");
      return res.json();
    },
  });

  const markRead = useMutation({
    mutationFn: async (id: number) => {
      await fetch(`/api/adviser/notifications/${id}/read`, {
        method: "PATCH",
      });
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["/api/adviser/notifications"] });
    },
  });

  const notifications = data?.notifications ?? [];
  const unreadCount = notifications.filter((n: any) => n.status === "unread").length;

  return (
    <div className="relative group">
      <Button variant="ghost" className="relative">
        <Bell className="h-5 w-5" />
        {unreadCount > 0 && (
          <span className="absolute -right-1 -top-1 rounded-full bg-red-500 px-1.5 text-xs text-white">
            {unreadCount}
          </span>
        )}
      </Button>

      <div className="absolute right-0 z-50 mt-2 hidden w-96 rounded-xl border bg-white p-3 shadow-xl group-hover:block">
        <div className="mb-2 font-semibold">Notifications</div>

        {notifications.length === 0 && (
          <div className="text-sm text-gray-500">No notifications.</div>
        )}

        {notifications.slice(0, 8).map((n: any) => (
          <button
            key={n.id}
            className="block w-full rounded-lg p-3 text-left hover:bg-gray-50"
            onClick={async () => {
              await markRead.mutateAsync(n.id);
              if (n.deepLink) navigate(n.deepLink);
            }}
          >
            <div className="flex justify-between gap-3">
              <div className="font-medium">{n.title}</div>
              <div className="text-xs text-gray-400">{n.priority}</div>
            </div>
            <div className="text-sm text-gray-500">{n.message}</div>
          </button>
        ))}
      </div>
    </div>
  );
}
