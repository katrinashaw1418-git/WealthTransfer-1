     must verify a valid adviceDisclaimerAcknowledgement exists
     for the requesting client and plan before returning any data.
   - If no acknowledgement: return 403 with { code: "DISCLAIMER_REQUIRED" }
   - Client must POST /api/client/wealth-plan/acknowledge-disclaimer first.
   - Disclaimer version must match the current active version in config.
   - If version mismatch: return 403 with { code: "DISCLAIMER_VERSION_OUTDATED" }

2. Role access rules
   - Adviser:
     can create and read plans for their own assigned clients only
     cannot read another adviser's plans
     cannot write to admin-only fields
     cannot set adviceType to anything other than "general"
   - Client:
     can read their own plan only (no write access to any plan table)
     cannot read planner notes with visibility = adviser_only or admin_only
   - Admin:
     can read all plans
     cannot mutate plan records directly
     can flag a plan for compliance review via POST /api/admin/wealth-plans/:id/flag
   - Compliance_admin:
     can read all plans and all notes
     can flag, can override review status
     cannot create or mutate plan records

3. Advice type lock
   - Every POST to /api/adviser/wealth-plans/:id/recommendations must
     reject any body where adviceType !== "general".
   - Return 400 with { code: "PERSONAL_ADVICE_BLOCKED",
     message: "Personal advice requires AFSL/AR framework. adviceType must be general." }
   - This is enforced server-side. UI selection is irrelevant.

4. Append-only enforcement
   - No PUT or PATCH on plannerNotes, riskProfileSnapshots, clientObjectives,
     adviceDisclaimerAcknowledgements, wealthPlanVersions, wealthPlanDocuments.
   - Only POST (new row) is permitted on these tables.
   - Objective updates: POST a new objective row, then set supersededAt on the old one
     in the same DB transaction.

5. Version on every plan update
   - Any write to wealthPlans, riskProfileSnapshots, clientObjectives, or
     wealthPlanRecommendations must atomically create a new wealthPlanVersions row
     with versionNumber = prior max + 1 and snapshotData = full plan state at that moment.

6. Audit log on every write
   - All plan writes must be recorded in the existing audit log with:
     userId, action, resourceType, resourceId, timestamp, ipAddress.

// services/wealthPlannerService.ts
//
// ALL writes to any planner table must go through this wrapper.
// Direct ORM writes outside this function are prohibited.
// assertWriteRole must be called at the top of every write path.

export async function writeWithVersioning(
  db: DbClient,
  planId: number,
  actorUserId: number,
  changeReason: string,
  isMaterialChange: boolean,
  writeFn: (tx: Transaction) => Promise<void>
): Promise<void> {
  await db.transaction(async (tx) => {
    // 1. Verify plan is not under_review before any non-compliance write
    //    (freeze endpoint is exempt from this check)
    const plan = await tx.query.wealthPlans.findFirst({
      where: eq(wealthPlans.id, planId)
    });
    if (!plan) throw new NotFoundError("wealth_plan_not_found");
    if (plan.status === "under_review") {
      throw new LockedError("plan_under_review");
    }

    // 2. Run the caller's write operations
    await writeFn(tx);

    // 3. Always create a version snapshot — no exceptions
    const prior = await tx.query.wealthPlanVersions.findFirst({
      where: eq(wealthPlanVersions.wealthPlanId, planId),
      orderBy: desc(wealthPlanVersions.versionNumber),
    });
    const nextVersion = (prior?.versionNumber ?? 0) + 1;
    const snapshot = await buildPlanSnapshot(tx, planId);

    await tx.insert(wealthPlanVersions).values({
      wealthPlanId: planId,
      versionNumber: nextVersion,
      snapshotData: snapshot,
      changeReason,
      isMaterialChange,
      createdBy: actorUserId,
    });

    // 4. Write audit log with before + after state
    await writeAuditLog(tx, {
      userId: actorUserId,
      action: changeReason,
      resourceType: "wealth_plan",
      resourceId: planId,
      metadata: {
        before: prior?.snapshotData ?? null,
        after: snapshot,
        versionNumber: nextVersion,
        isMaterialChange,
      },
    });
  });
}

// Role assertion — called at top of every write path, no exceptions
export function assertWriteRole(user: User): void {
  if (user.role !== "adviser") {
    throw new ForbiddenError("write_role_required");
  }
}

// Role assertion for reads — client sees own plan only, adviser sees own clients
export function assertReadAccess(user: User, plan: WealthPlan): void {
  if (user.role === "client" && plan.clientUserId !== user.id) {
    throw new ForbiddenError("plan_access_denied");

// middleware/disclaimerGate.ts
//
// Applied to: GET /api/client/wealth-plan and all client plan data routes.
// Checks both: disclaimerVersion AND wealthPlanVersionId staleness.

export async function requireDisclaimerAcknowledgement(
  req: Request,
  res: Response,
  next: NextFunction
) {
  const { planId, userId } = req;

  // Get current active disclaimer version from config
  const activeDisclaimerVersion = config.ACTIVE_DISCLAIMER_VERSION;

  // Get the most recent material change version for this plan
  const lastMaterialVersion = await db.query.wealthPlanVersions.findFirst({
    where: and(
      eq(wealthPlanVersions.wealthPlanId, planId),
      eq(wealthPlanVersions.isMaterialChange, true)
    ),
    orderBy: desc(wealthPlanVersions.versionNumber),
  });

  // Get the client's most recent acknowledgement for this plan
  const ack = await db.query.adviceDisclaimerAcknowledgements.findFirst({
    where: and(
      eq(adviceDisclaimerAcknowledgements.clientUserId, userId),
      eq(adviceDisclaimerAcknowledgements.wealthPlanId, planId)
    ),
    orderBy: desc(adviceDisclaimerAcknowledgements.acknowledgedAt),
  });

  // Gate 1: no acknowledgement at all
  if (!ack) {
    return res.status(403).json({ code: "DISCLAIMER_REQUIRED" });
  }

  // Gate 2: disclaimer text version is outdated
  if (ack.disclaimerVersion !== activeDisclaimerVersion) {
    return res.status(403).json({ code: "DISCLAIMER_VERSION_OUTDATED" });
  }

  // Gate 3: a material plan change occurred after the client's last acknowledgement
  // The ack is stale if any material version was created after it was acknowledged
  if (lastMaterialVersion) {
    const ackVersion = await db.query.wealthPlanVersions.findFirst({
      where: eq(wealthPlanVersions.id, ack.wealthPlanVersionId)
    });
    if (!ackVersion || ackVersion.versionNumber < lastMaterialVersion.versionNumber) {
      return res.status(403).json({ code: "DISCLAIMER_STALE_AFTER_MATERIAL_CHANGE" });
    }
  }

  next();
}
