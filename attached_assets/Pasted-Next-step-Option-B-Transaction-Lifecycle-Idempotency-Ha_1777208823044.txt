2. Add lifecycle helper

Create:

server/services/transaction-lifecycle.ts
const ALLOWED_TRANSITIONS: Record<string, string[]> = {
  pending: ["settled", "failed"],
  settled: ["reversed"],
  failed: [],
  reversed: [],
};

export function assertTransactionTransition(from: string, to: string) {
  if (!ALLOWED_TRANSITIONS[from]?.includes(to)) {
    throw new Error(`Invalid transaction transition: ${from} → ${to}`);
  }
}
3. Add idempotency helper
export async function findOrCreateIdempotentTransaction({
  userId,
  idempotencyKey,
  create,
}: {
  userId: number;
  idempotencyKey: string;
  create: () => Promise<any>;
}) {
  const existing = await db
    .select()
    .from(transactions)
    .where(and(
      eq(transactions.userId, userId),
      eq(transactions.idempotencyKey, idempotencyKey)
    ));

  if (existing.length > 0) {
    return { transaction: existing[0], idempotentReplay: true };
  }

  const transaction = await create();
  return { transaction, idempotentReplay: false };
}

    .returning();

  await tx.insert(ledgerEntries).values({

    amount: updated.amount,
    direction: updated.type === "deposit" ? "credit" : "debit",
    entryType: `${updated.type}_settlement`,
    referenceType: "transaction",
