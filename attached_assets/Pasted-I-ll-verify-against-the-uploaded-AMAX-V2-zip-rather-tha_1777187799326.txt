async function getRole(userId: number): Promise<string> {
  const [user] = await db
    .select({ role: users.role })
    .from(users)
    .where(eq(users.id, userId));

  return user?.role ?? "client";
}

async function adviserCanAccessClient(adviserUserId: number, clientUserId: number) {
  const [link] = await db
    .select()
    .from(adviserClients)
    .where(and(
      eq(adviserClients.adviserUserId, adviserUserId),
      eq(adviserClients.clientUserId, clientUserId),
    ));

  return !!link;
}
