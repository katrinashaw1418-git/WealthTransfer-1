## #94 — service wrapper

**Create `server/services/adviceRecordService.ts`:**

```ts
import { db } from "../db";
import { eq, desc } from "drizzle-orm";
import { adviceRecords, adviceRecordVersions, auditLogs } from "../../shared/schema";

export async function updateAdviceRecordStatus(
  adviceRecordId: number,
  newStatus: string,
  changeReason: string,
  actorUserId: number,
  ipAddress?: string
): Promise<void> {
  await db.transaction(async (tx) => {
    const record = await tx.query.adviceRecords.findFirst({
      where: eq(adviceRecords.id, adviceRecordId),
    });
    if (!record) throw new Error(`advice_record_not_found:${adviceRecordId}`);

    if (
      record.status === "review_pending" &&
      newStatus !== "active" &&
      newStatus !== "closed"
    ) {
      throw new Error("advice_record_locked:review_pending");
    }

    const before = { ...record };

    await tx.update(adviceRecords)
      .set({ status: newStatus })
      .where(eq(adviceRecords.id, adviceRecordId));

    const latest = await tx.query.adviceRecordVersions.findFirst({
      where: eq(adviceRecordVersions.adviceRecordId, adviceRecordId),
      orderBy: desc(adviceRecordVersions.versionNumber),
    });
    const nextVersion = (latest?.versionNumber ?? 0) + 1;

    await tx.insert(adviceRecordVersions).values({
      adviceRecordId,
      versionNumber: nextVersion,
      snapshotJsonb: { ...record, status: newStatus },
      changeReason,
      createdBy: actorUserId,
    });

    await tx.insert(auditLogs).values({
      userId: actorUserId,
      action: `advice_record_status:${record.status}→${newStatus}`,
      resourceType: "advice_record",
      resourceId: String(adviceRecordId),
      metadata: {
        before,
        after: { ...record, status: newStatus },
        versionNumber: nextVersion,
        changeReason,
        ipAddress: ipAddress ?? null,
      },
    });
  });
}

export async function assertNotUnderReview(adviceRecordId: number): Promise<void> {
  const record = await db.query.adviceRecords.findFirst({
    where: eq(adviceRecords.id, adviceRecordId),
  });
  if (!record) throw new Error(`advice_record_not_found:${adviceRecordId}`);
  if (record.status === "review_pending") {
    throw new Error("advice_record_locked:review_pending");
  }
}
```

## #94 — route guard

**Create `server/middleware/adviceRecordGuards.ts`:**

```ts
import { Request, Response, NextFunction } from "express";
import { assertNotUnderReview } from "../services/adviceRecordService";

export async function blockIfUnderReview(
  req: Request,
  res: Response,
  next: NextFunction
) {
  const id = parseInt(req.params.id ?? req.params.adviceRecordId);
  if (isNaN(id)) return next();

  try {
    await assertNotUnderReview(id);
    next();
  } catch (err: any) {
    if (err.message?.includes("review_pending")) {
      return res.status(423).json({
        code: "ADVICE_RECORD_LOCKED",
        message: "This record is under compliance review. Writes are blocked.",
      });
    }
    next(err);
  }
}
```

**Wire into routes:**

```ts
import { blockIfUnderReview } from "./middleware/adviceRecordGuards";

// Apply to objectives and recommendations — NOT to notes
router.post("/api/adviser/advice-records/:id/objectives",
  requireAuth, blockIfUnderReview, objectivesHandler);

router.post("/api/adviser/advice-records/:id/recommendations",
  requireAuth, blockIfUnderReview, recommendationsHandler);

// Notes exempt — no blockIfUnderReview here
router.post("/api/adviser/advice-records/:id/notes",
  requireAuth, notesHandler);
```
