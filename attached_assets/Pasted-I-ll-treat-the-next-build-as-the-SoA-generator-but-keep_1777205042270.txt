6. Backend skeleton
app.post("/api/adviser/soa-drafts", async (req, res) => {
  const { userId } = requireAuth(req);
  const role = await getRole(userId);

  if (role !== "adviser") {
    return res.status(403).json({ error: "Adviser access required" });
  }

  const {
    clientUserId,
    instructionId,
    scope,
    basisOfAdvice,
    recommendationSummary,
    risks,
    feesAndCosts,
    conflictsDisclosure,
    alternativesConsidered,
    replacementProductDisclosure,
  } = req.body;

  const allowed = await adviserCanAccessClient(userId, clientUserId);
  if (!allowed) {
    return res.status(403).json({ error: "Client not linked to adviser" });
  }

  const documentHtml = renderSoaHtml({
    scope,
    basisOfAdvice,
    recommendationSummary,
    risks,
    feesAndCosts,
    conflictsDisclosure,
    alternativesConsidered,
    replacementProductDisclosure,
  });

  const documentHash = createHash("sha256").update(documentHtml).digest("hex");

  const [soa] = await db.insert(soaDocuments).values({
    adviserUserId: userId,
    clientUserId,
    instructionId,
    scope,
    basisOfAdvice,
    recommendationSummary,
    risks,
    feesAndCosts,
    conflictsDisclosure,
    alternativesConsidered,
    replacementProductDisclosure,
    documentHtml,
    documentHash,
    documentStatus: "draft",
  }).returning();

  await writeAuditLog(
    userId,
    "soa_draft_created",
    "soa_documents",
    String(soa.id),
    { clientUserId, instructionId },
    req.ip || null
  );

  res.status(201).json({ soa });
});
