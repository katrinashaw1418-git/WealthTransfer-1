export function calculateDailyPercentageFee(baseAmount: number, rateBps: number) {
  return (baseAmount * (rateBps / 10000)) / 365;
}

export function splitFee(gross: number, adviserSplitBps: number, platformSplitBps: number) {
  if (adviserSplitBps + platformSplitBps !== 10000) {
    throw new Error("Fee splits must equal 10000 bps");
  }

  return {
    adviserAmount: gross * adviserSplitBps / 10000,
    platformAmount: gross * platformSplitBps / 10000,
  };
}

Base amount:

Use current portfolio/wallet-derived AUM.
If unavailable, baseAmount = 0 and grossFeeAmount = 0.
Do not use fake history or projected performance.

5. Run accruals

Process:

For each active adviserFeeRule:
1. Load feeConsent
2. Validate active/not expired/not withdrawn
3. Validate adviserClients link
4. Calculate base AUM
5. Calculate gross fee
6. Split adviser/platform
7. Insert one accrual per feeRuleId + accrualDate
8. Create pending deduction record

Response:

{

  "createdAccruals": 8,
  "createdDeductions": 8,
  "skipped": 2
