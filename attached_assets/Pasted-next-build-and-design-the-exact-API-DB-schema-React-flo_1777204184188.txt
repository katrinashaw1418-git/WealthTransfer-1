export async function sendRegistrationInviteEmail({
  email,
  role,
  inviteLink,
}: {
  email: string;
  role: string;
  inviteLink: string;
}) {
  const subject = "Activate your AMAX Wealth account";

  const body = `
You have been invited to activate your AMAX Wealth ${role} account.

Activate your account:
${inviteLink}

This link expires in 48 hours and can only be used once.

If you did not expect this invite, please ignore this email.
`;

  return sendEmail({
    to: email,
    subject,
    text: body,
  });
}
4. Create invite endpoint patch
let emailSent = false;
let emailError: string | null = null;

try {
  await sendRegistrationInviteEmail({
    email: normalizedEmail,
    role: inviteRole,
    inviteLink,
  });

  emailSent = true;

  await db.update(registrationInvites)
    .set({
      emailSentAt: new Date(),
      emailSendStatus: "sent",
      emailError: null,
    })
    .where(eq(registrationInvites.id, invite.id));

} catch (err: any) {
  emailError = err.message;

  await db.update(registrationInvites)
    .set({
      emailSendStatus: "failed",
      emailError,
    })
    .where(eq(registrationInvites.id, invite.id));
}

Return:

res.json({

  emailSent,
  emailError,
