## B. Reconciliation service

```ts
export async function runReconciliation() {
  const usersWithBalances = await db
    .select({
      userId: ledgerEntries.userId,
      currency: ledgerEntries.currency,
    })
    .from(ledgerEntries)
    .groupBy(ledgerEntries.userId, ledgerEntries.currency);

  for (const row of usersWithBalances) {
    const internalBalance = await getUserCurrencyBalance(
      row.userId,
      row.currency
    );

    // TODO: replace with real custodian API
    const externalBalance = await fetchCustodianBalance(
      row.userId,
      row.currency
    );

    const diff =
      Number(internalBalance) - Number(externalBalance);

    await db.insert(reconciliations).values({
      userId: row.userId,
      currency: row.currency,
      internalBalance,
      externalBalance,
      difference: diff.toFixed(8),
      status: Math.abs(diff) < 0.01 ? "match" : "mismatch",
    });

    if (Math.abs(diff) > 100) {
      console.error("RECONCILIATION ALERT", {
        userId: row.userId,
        currency: row.currency,
        diff,
      });
    }
  }
}
```
