1. DB schema (minimal, safe)
export const adviserNotificationDismissals = pgTable(
  "adviser_notification_dismissals",
  {
    id: serial("id").primaryKey(),

    adviserUserId: integer("adviser_user_id")
      .references(() => users.id)
      .notNull(),

    sourceType: text("source_type").notNull(),
    // instruction | task | fee_consent | report | kyc

    sourceId: integer("source_id").notNull(),

    dismissedAt: timestamp("dismissed_at").defaultNow(),
  }
);

👉 This does NOT duplicate notifications
👉 It only stores “user preference”

2. API (very small)

app.post("/api/adviser/notifications/dismiss", async (req, res) => {
  const { userId } = requireAuth(req);
  const { sourceType, sourceId } = req.body;

  await db.insert(adviserNotificationDismissals).values({
    adviserUserId: userId,
    sourceType,
    sourceId,
  });

  res.json({ success: true });
});
Undo dismiss
app.delete("/api/adviser/notifications/dismiss", async (req, res) => {
  const { userId } = requireAuth(req);
  const { sourceType, sourceId } = req.body;

  await db
    .delete(adviserNotificationDismissals)
    .where(and(
      eq(adviserNotificationDismissals.adviserUserId, userId),
      eq(adviserNotificationDismissals.sourceType, sourceType),
      eq(adviserNotificationDismissals.sourceId, sourceId)
    ));

  res.json({ success: true });
});
