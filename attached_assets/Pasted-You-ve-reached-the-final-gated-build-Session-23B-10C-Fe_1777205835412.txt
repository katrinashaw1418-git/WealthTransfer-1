Deduct endpoint rules

Before any deduction:

1. admin/compliance_admin only
2. deduction status must be approved
3. linked fee consent must be active
4. consent must not be expired
5. consent must not be withdrawn
6. client wallet must have sufficient available balance
7. deduction must not already have a posted journal
8. all wallet changes must be atomic transaction
9. all writes must audit log
Deduct API skeleton
app.post("/api/admin/fee-deductions/:id/deduct", async (req, res) => {
  const { userId } = requireAuth(req);
  const role = await getRole(userId);

  if (!["admin", "compliance_admin"].includes(role)) {
    return res.status(403).json({ error: "Admin access required" });
  }

  const deductionId = Number(req.params.id);

  return await db.transaction(async (tx) => {
    const [deduction] = await tx
      .select()
      .from(adviserFeeDeductions)
      .where(eq(adviserFeeDeductions.id, deductionId));

    if (!deduction) throw new Error("Deduction not found");

    if (deduction.status !== "approved") {
      return res.status(400).json({ error: "Deduction must be approved before deduction" });
    }

    const [consent] = await tx
      .select()
      .from(feeConsents)
      .where(eq(feeConsents.id, deduction.feeConsentId));

    if (!consent || consent.renewalStatus !== "active") {
      return res.status(400).json({ error: "Active fee consent required" });
    }

    if (consent.withdrawnAt) {
      return res.status(400).json({ error: "Fee consent has been withdrawn" });
    }

    if (new Date() > new Date(consent.consentExpiryDate)) {
      return res.status(400).json({ error: "Fee consent has expired" });
    }

    const [existingJournal] = await tx
      .select()
      .from(feeDeductionJournals)
      .where(eq(feeDeductionJournals.deductionId, deductionId));

    if (existingJournal) {
      return res.status(409).json({ error: "Deduction already posted" });
    }

    const [clientWallet] = await tx
      .select()
      .from(wallets)
      .where(and(
        eq(wallets.userId, deduction.clientUserId),
        eq(wallets.currency, deduction.currency)
      ));

    if (!clientWallet) {
      return res.status(400).json({ error: "Client wallet not found" });
    }

    if (Number(clientWallet.availableBalance) < Number(deduction.grossAmount)) {
      return res.status(400).json({ error: "Insufficient available balance" });
    }

    // Debit client wallet
    await tx.update(wallets)
      .set({
        balance: sql`${wallets.balance} - ${deduction.grossAmount}`,
        availableBalance: sql`${wallets.availableBalance} - ${deduction.grossAmount}`,
      })
      .where(eq(wallets.id, clientWallet.id));

    // For first version, journal split only.
    // Actual adviser/platform wallet credits can remain reconciliation-only if platform wallets are not finalised.

    const [journal] = await tx.insert(feeDeductionJournals).values({
      deductionId,
      clientUserId: deduction.clientUserId,
      adviserUserId: deduction.adviserUserId,
      debitWalletId: clientWallet.id,
      grossAmount: deduction.grossAmount,
      adviserAmount: deduction.adviserAmount,
      platformAmount: deduction.platformAmount,
      currency: deduction.currency,
      status: "posted",
      postedBy: userId,
    }).returning();

    const [updated] = await tx.update(adviserFeeDeductions)
      .set({
        status: "deducted",
        deductedAt: new Date(),
      })
      .where(eq(adviserFeeDeductions.id, deductionId))
      .returning();

    await writeAuditLog(
      userId,
      "fee_deduction_posted",
      "adviser_fee_deductions",
      String(deductionId),
      { deductionId, journalId: journal.id, grossAmount: deduction.grossAmount },
      req.ip || null
    );

    return res.json({ deduction: updated, journal });
  });
});
