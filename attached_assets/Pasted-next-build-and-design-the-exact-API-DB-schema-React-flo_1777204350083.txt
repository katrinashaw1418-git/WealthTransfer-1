1. DB schema

Add:

export const adviserNotificationDismissals = pgTable("adviser_notification_dismissals", {
  id: serial("id").primaryKey(),

  adviserUserId: integer("adviser_user_id")
    .references(() => users.id)
    .notNull(),

  sourceType: text("source_type").notNull(),
  // instruction | task | fee_consent | report | kyc

  sourceId: integer("source_id").notNull(),

  dismissedAt: timestamp("dismissed_at").defaultNow(),
}, (table) => ({
  uniq: uniqueIndex("adviser_notification_dismissals_uniq")
    .on(table.adviserUserId, table.sourceType, table.sourceId),
}));

app.post("/api/adviser/notifications/dismiss", async (req, res) => {
  const { userId } = requireAuth(req);
  const role = await getRole(userId);

  if (role !== "adviser") {
    return res.status(403).json({ error: "Adviser access required" });
  }

  const { sourceType, sourceId } = req.body;

  if (!sourceType || !sourceId) {
    return res.status(400).json({ error: "sourceType and sourceId required" });
  }

  await db.insert(adviserNotificationDismissals)
    .values({
      adviserUserId: userId,
      sourceType,
      sourceId: Number(sourceId),
    })
    .onConflictDoNothing();

  res.json({ success: true });
});
Undo dismiss
app.delete("/api/adviser/notifications/dismiss", async (req, res) => {
  const { userId } = requireAuth(req);
  const role = await getRole(userId);

  if (role !== "adviser") {
    return res.status(403).json({ error: "Adviser access required" });
  }

  const { sourceType, sourceId } = req.body;

  await db.delete(adviserNotificationDismissals)
    .where(and(
      eq(adviserNotificationDismissals.adviserUserId, userId),
      eq(adviserNotificationDismissals.sourceType, sourceType),
      eq(adviserNotificationDismissals.sourceId, Number(sourceId))
    ));

  res.json({ success: true });
});

const dismissedSet = new Set(
