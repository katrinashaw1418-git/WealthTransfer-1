app.post("/api/adviser/fee-consents", async (req, res) => {
  const { userId } = requireAuth(req);
  const role = await getRole(userId);

  if (role !== "adviser") {
    return res.status(403).json({ error: "Adviser access required" });
  }

  const {
    clientUserId,
    accountNumber,
    accountName,
    feeType,
    amountType,
    amount,
    referenceDay,
    services,
  } = req.body;

  const allowed = await adviserCanAccessClient(userId, clientUserId);
  if (!allowed) {
    return res.status(403).json({ error: "Client not linked to adviser" });
  }

  const ref = new Date(referenceDay);
  const renewalWindowStart = new Date(ref.getTime() - 60 * 24 * 60 * 60 * 1000);
  const renewalWindowEnd = new Date(ref.getTime() + 150 * 24 * 60 * 60 * 1000);

  const [consent] = await db.insert(feeConsents).values({
    adviserUserId: userId,
    clientUserId,
    accountNumber,
    accountName,
    feeType,
    amountType,
    amount: String(amount),
    referenceDay: ref,
    renewalWindowStart,
    renewalWindowEnd,
    consentExpiryDate: renewalWindowEnd,
    renewalStatus: "pending_client_signature",
    services,
  }).returning();

  await writeAuditLog(
    userId,
    "fee_consent_created",
    "fee_consents",
    String(consent.id),
    { clientUserId, feeType, amountType, amount },
    req.ip || null
  );

  res.status(201).json({ consent });
});
5. Client sign API
POST /api/client/fee-consents/:id/sign

Body:

{
  "clientSignatureName": "Wise User"
}

Rules:

clientUserId must equal logged-in client
status becomes active
clientSignedAt set
no fee deduction
audit log
app.post("/api/client/fee-consents/:id/sign", async (req, res) => {
  const { userId } = requireAuth(req);
  const id = Number(req.params.id);

  const [consent] = await db.select().from(feeConsents)
    .where(eq(feeConsents.id, id));

  if (!consent || consent.clientUserId !== userId) {
    return res.status(404).json({ error: "Fee consent not found" });
  }

  const [updated] = await db.update(feeConsents)
    .set({
      renewalStatus: "active",
      clientSignatureName: req.body.clientSignatureName,
      clientSignedAt: new Date(),
    })
    .where(eq(feeConsents.id, id))
    .returning();

  await writeAuditLog(
    userId,
    "fee_consent_signed",
    "fee_consents",
    String(id),
    { clientSignatureName: req.body.clientSignatureName },
    req.ip || null
  );

  res.json({ consent: updated });
});

7. React adviser form skeleton
function CreateFeeConsentForm() {
  const [form, setForm] = useState({
    clientUserId: "",
    accountNumber: "",
    accountName: "",
    feeType: "ongoing_service",
    amountType: "percentage",
    amount: "",
    referenceDay: "",
    services: "",
  });

  async function submit() {
    const res = await fetch("/api/adviser/fee-consents", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        ...form,
        clientUserId: Number(form.clientUserId),
      }),
    });

    if (!res.ok) {
      const data = await res.json();
      alert(data.error || "Failed to create fee consent");
      return;
    }

    alert("Fee consent sent to client for signature.");
  }

  return (
    <div className="space-y-3">
      <input placeholder="Client user ID" />
      <input placeholder="Account number" />
      <input placeholder="Account name" />
      <select>
        <option value="ongoing_service">Ongoing service</option>
        <option value="one_off_advice">One-off advice</option>
      </select>
      <select>
        <option value="percentage">Percentage</option>
        <option value="fixed">Fixed amount</option>
      </select>
      <input placeholder="Amount" />
      <input type="date" />
      <textarea placeholder="Services covered" />
      <button onClick={submit}>Create fee consent</button>
    </div>
  );
}
8. React client sign flow
async function signConsent(id: number, clientSignatureName: string) {
  const res = await fetch(`/api/client/fee-consents/${id}/sign`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ clientSignatureName }),
  });

  if (!res.ok) {
    const data = await res.json();
    throw new Error(data.error || "Failed to sign consent");
  }
}
