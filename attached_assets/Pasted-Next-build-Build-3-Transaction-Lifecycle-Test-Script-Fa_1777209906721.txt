1. scripts/test-transaction-safety.ts
type TestResult = {
  name: string;
  passed: boolean;
  details?: string;
};

const results: TestResult[] = [];

function pass(name: string, details?: string) {
  results.push({ name, passed: true, details });
}

function fail(name: string, details?: string) {
  results.push({ name, passed: false, details });
}

async function run() {
  // 1. deposit idempotency
  // create same money request 3 times with same Idempotency-Key
  // assert only one transaction exists

  // 2. pending no ledger
  // create pending transaction
  // assert zero ledger entries for transaction

  // 3. settlement retry
  // settle same transaction twice
  // assert exactly one ledger entry

  // 4. failed no ledger
  // fail pending transaction
  // assert zero ledger entries

  // 5. reversal offset
  // reverse settled transaction
  // assert one opposite-direction ledger entry

  // 6. reconciliation mismatch
  // create artificial cached wallet mismatch
  // run reconciliation
  // assert status = mismatch

  for (const r of results) {
    console.log(`${r.passed ? "PASS" : "FAIL"} ${r.name}${r.details ? ` — ${r.details}` : ""}`);
  }

  const failed = results.filter(r => !r.passed);

  if (failed.length > 0) {
    console.error(`\n${failed.length} transaction safety test(s) failed.`);
    process.exit(1);
  }

  console.log("\nALL TRANSACTION SAFETY TESTS PASSED ✅");
}

run().catch((err) => {
  console.error("Transaction safety test crashed:", err);
  process.exit(1);
});
