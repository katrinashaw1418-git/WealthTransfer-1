1. Database table
export const registrationInvites = pgTable("registration_invites", {
  id: serial("id").primaryKey(),

  email: text("email").notNull(),
  role: text("role").notNull(), 
  // client | adviser | admin

  relatedEntityType: text("related_entity_type"),
  // adviser_application | client_application | adviser_firm

  relatedEntityId: integer("related_entity_id"),

  adviserUserId: integer("adviser_user_id").references(() => users.id),

  inviteHash: text("invite_hash").notNull().unique(),

  expiresAt: timestamp("expires_at").notNull(),
  usedAt: timestamp("used_at"),

  createdBy: integer("created_by").references(() => users.id),
  createdAt: timestamp("created_at").defaultNow(),
});

Use invite, not “token”, if the wording bothers you.

2. Admin creates invite
POST /api/admin/registration-invites

Body:

{
  "email": "planner@example.com",
  "role": "adviser",
  "relatedEntityType": "adviser_application",
  "relatedEntityId": 123
}

Server:

const rawInvite = crypto.randomBytes(32).toString("hex");
const inviteHash = hash(rawInvite);

await db.insert(registrationInvites).values({
  email,
  role,
  relatedEntityType,
  relatedEntityId,
  inviteHash,
  expiresAt: new Date(Date.now() + 48 * 60 * 60 * 1000),
  createdBy: adminUserId,
});

return {
  inviteLink: `${APP_URL}/register/invite?invite=${rawInvite}`
};

5. Complete registration
POST /api/auth/registration-invites/complete

Body:

{
  "invite": "rawInviteValue",
  "password": "StrongPassword123!"

Server creates user:

const user = await storage.createUser({
  email: invite.email,
  username: invite.email,
  password: await hashPassword(password),
  role: invite.role,
  emailVerified: true,
  kycStatus: invite.role === "client" ? "pending" : "not_required",
});

Then:

await db.update(registrationInvites)
  .set({ usedAt: new Date() })
  .where(eq(registrationInvites.id, invite.id));

await db.insert(adviserClients).values({
