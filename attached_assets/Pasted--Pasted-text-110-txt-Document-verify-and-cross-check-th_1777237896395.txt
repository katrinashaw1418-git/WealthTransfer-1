🧠 Patch 10 — version wrapper (critical)
export async function writeWithVersioning(
  db: any,
  adviceRecordId: number,
  userId: number,
  changeReason: string,
  writeFn: (tx: any) => Promise<void>
) {
  await db.transaction(async (tx: any) => {
    const before = await tx.query.adviceRecords.findFirst({
      where: eq(adviceRecords.id, adviceRecordId),
    });

    await writeFn(tx);

    const after = await tx.query.adviceRecords.findFirst({
      where: eq(adviceRecords.id, adviceRecordId),
    });

    const lastVersion = await tx.query.adviceRecordVersions.findFirst({
      where: eq(adviceRecordVersions.adviceRecordId, adviceRecordId),
      orderBy: desc(adviceRecordVersions.versionNumber),
    });

    const nextVersion = (lastVersion?.versionNumber ?? 0) + 1;

// server/services/adviceVersioning.ts

export async function writeAdviceWithVersioning(
  db: any,
  adviceRecordId: number,
  userId: number,
  changeReason: string,
  isMaterialChange: boolean,
  writeFn: (tx: any) => Promise<void>,
) {
  await db.transaction(async (tx: any) => {
    const before = await tx.query.adviceRecords.findFirst({
      where: eq(adviceRecords.id, adviceRecordId),
    });

    if (!before) throw new Error("ADVICE_RECORD_NOT_FOUND");

    if (isMaterialChange) {
      assertNotReviewPending(before);
    }

    await writeFn(tx);

    const after = await tx.query.adviceRecords.findFirst({
      where: eq(adviceRecords.id, adviceRecordId),
    });

    const lastVersion = await tx.query.adviceRecordVersions.findFirst({
      where: eq(adviceRecordVersions.adviceRecordId, adviceRecordId),
      orderBy: desc(adviceRecordVersions.versionNumber),
    });

    await tx.insert(adviceRecordVersions).values({
      adviceRecordId,
      versionNumber: (lastVersion?.versionNumber ?? 0) + 1,
      snapshotData: after,
      changeReason,
      isMaterialChange,
      createdBy: userId,
    });

    await writeStructuredAuditLog(tx, {
      userId,
      action: changeReason,
      resourceType: "advice_record",
      resourceId: adviceRecordId,
      before,
      after,
    });
  });
}
