function StatusBadge({ value }: { value: string }) {
  const style =
    value.toLowerCase().includes("verified") || value.toLowerCase().includes("active") || value.toLowerCase().includes("consented")
      ? "bg-emerald-500/10 text-emerald-300 border-emerald-500/20"
      : value.toLowerCase().includes("pending") || value.toLowerCase().includes("expiring")
      ? "bg-amber-500/10 text-amber-300 border-amber-500/20"
      : "bg-slate-500/10 text-slate-300 border-slate-500/20";

  return <Badge className={`border ${style}`}>{value}</Badge>;
}

export default function AMAXAdviserPortalDemo() {
  const [active, setActive] = useState("dashboard");
  const [selectedClient, setSelectedClient] = useState(clients[0]);

  const screen = useMemo(() => {
    switch (active) {
      case "dashboard": return <DashboardScreen setActive={setActive} />;
      case "clients": return <ClientsScreen setActive={setActive} setSelectedClient={setSelectedClient} />;
      case "client-detail": return <ClientDetailScreen client={selectedClient} />;
      case "products": return <ProductsScreen />;
      case "instructions": return <InstructionsScreen />;
      case "fees": return <FeesScreen />;
      case "tasks": return <TasksScreen />;
      case "reports": return <ReportsScreen />;
      default: return <DashboardScreen setActive={setActive} />;
    }
  }, [active, selectedClient]);

  return (
    <div className={`min-h-screen ${navy} text-white`}>
      <div className="border-b border-white/10 p-6">
        <AdviserLandingSection setActive={setActive} />
      </div>
      <div className="flex min-h-screen">
        <AdviserSidebar active={active} setActive={setActive} />

          <TopBar />
          <div className="p-5 md:p-8">{screen}</div>
        </main>
      </div>
    </div>
  );
}
