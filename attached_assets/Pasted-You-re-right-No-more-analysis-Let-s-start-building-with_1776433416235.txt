# ⚙️ 2. FRONTEND (React example)

```jsx
import { useState } from "react";

export default function EmailCapture({ profileData }) {
  const [email, setEmail] = useState("");
  const [status, setStatus] = useState("idle"); // idle | loading | success | error

  const handleSubmit = async () => {
    if (!email) return;

    setStatus("loading");

    try {
      await fetch("/api/email-capture", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          email,
          profile: profileData, // include recommendation output
        }),
      });

      setStatus("success");
    } catch (e) {
      setStatus("error");
    }
  };

  return (
    <div className="email-capture">
      <h3>Want a copy of this profile?</h3>

      <div style={{ display: "flex", gap: "8px" }}>
        <input
          type="email"
          placeholder="Enter your email"
          value={email}
          onChange={(e) => setEmail(e.target.value)}
        />

        <button onClick={handleSubmit} disabled={status === "loading"}>
          Email me this summary
        </button>
      </div>

      {status === "success" && <p>✓ Sent. Check your inbox.</p>}
      {status === "error" && <p>Something went wrong. Try again.</p>}
    </div>
  );
}
```

# 🧱 3. BACKEND (Node / Express example)

app.post("/api/email-capture", async (req, res) => {
  const { email, profile } = req.body;

  if (!email) {
    return res.status(400).json({ error: "Email required" });
  }

  // 1. Store lead
  await db.collection("leads").insertOne({
    email,
    profile,
    createdAt: new Date(),
  });

  // 2. Send email (example with nodemailer)
  await sendEmail({
    to: email,
    subject: "Your AMAX Wealth Profile",
    html: `
      <h2>Your Investment Profile</h2>
      <p><strong>Profile:</strong> ${profile.name}</p>
      <p><strong>Strategy:</strong> ${profile.strategy}</p>
      <p>${profile.notes}</p>

      <hr/>

      <p>Create your account to access opportunities:</p>

      <p style="font-size:12px;color:gray;">
        This is a general indication and does not constitute financial advice.
      </p>
    `,
  });

  res.json({ success: true });
});
