export async function getTransparentBalance(userId: number, currency: string) {
  const ledgerBalance = await getLedgerBalance(userId, currency);

  const [wallet] = await db
    .select()
    .from(wallets)
    .where(and(
      eq(wallets.userId, userId),
      eq(wallets.currency, currency)
    ));

  const cachedBalance = wallet ? Number(wallet.balance) : 0;
  const drift = ledgerBalance - cachedBalance;

  return {
    balance: ledgerBalance.toFixed(2),
    availableBalance: ledgerBalance.toFixed(2),
    balanceSource: "ledger",
    cachedBalance: cachedBalance.toFixed(2),
    driftAmount: drift.toFixed(2),
    hasBalanceMismatch: Math.abs(drift) > 0.01,
  };
}

2. server/services/reconciliation.ts
export async function runWalletReconciliation(userId: number, currency = "AUD") {
  const ledgerBalance = await getLedgerBalance(userId, currency);

  const [wallet] = await db
    .select()
    .from(wallets)
    .where(and(
      eq(wallets.userId, userId),
      eq(wallets.currency, currency)
    ));

  const cachedWalletBalance = wallet ? Number(wallet.balance) : 0;
  const driftAmount = ledgerBalance - cachedWalletBalance;

  const [report] = await db.insert(reconciliationReports).values({
    userId,
    currency,
    ledgerBalance: ledgerBalance.toFixed(2),
    cachedWalletBalance: cachedWalletBalance.toFixed(2),
    driftAmount: driftAmount.toFixed(2),
    status: Math.abs(driftAmount) > 0.01 ? "mismatch" : "matched",
  }).returning();

  return report;
}

app.post("/api/admin/reconciliation/run", async (req, res) => {
  const { userId } = requireAuth(req);
  const role = await getRole(userId);

  if (!["admin", "compliance_admin"].includes(role)) {
    return res.status(403).json({ error: "Admin access required" });
  }

  const allWallets = await db.select().from(wallets);
  const reports = [];

  for (const wallet of allWallets) {
    const report = await runWalletReconciliation(wallet.userId, wallet.currency);
    reports.push(report);
  }

  await writeAuditLog(
    userId,
    "reconciliation_run",
    "reconciliation_reports",
    null,
    { count: reports.length },
    req.ip || null
  );

  res.json({
    count: reports.length,
    mismatches: reports.filter(r => r.status === "mismatch").length,
    reports,
  });
});
