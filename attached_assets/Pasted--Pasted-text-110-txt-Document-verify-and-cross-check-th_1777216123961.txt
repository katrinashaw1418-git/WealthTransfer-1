🧠 Patch 10 — version wrapper (critical)
export async function writeWithVersioning(
  db: any,
  adviceRecordId: number,
  userId: number,
  changeReason: string,
  writeFn: (tx: any) => Promise<void>
) {
  await db.transaction(async (tx: any) => {
    const before = await tx.query.adviceRecords.findFirst({
      where: eq(adviceRecords.id, adviceRecordId),
    });

    await writeFn(tx);

    const after = await tx.query.adviceRecords.findFirst({
      where: eq(adviceRecords.id, adviceRecordId),
    });

    const lastVersion = await tx.query.adviceRecordVersions.findFirst({
      where: eq(adviceRecordVersions.adviceRecordId, adviceRecordId),
      orderBy: desc(adviceRecordVersions.versionNumber),
    });

    const nextVersion = (lastVersion?.versionNumber ?? 0) + 1;

    await tx.insert(adviceRecordVersions).values({
      adviceRecordId,
      versionNumber: nextVersion,
      snapshotData: after,
      changeReason,
      createdBy: userId,
    });

    await writeAuditLog(tx, {
      userId,
      action: changeReason,
      resourceType: "advice_record",
      resourceId: adviceRecordId,
      before,
      after,
    });
  });

function requireAdviser(req: any) {
  if (req.user?.role !== "adviser") {
    const err: any = new Error("ADVISER_REQUIRED");
    err.status = 403;
    throw err;
  }
}

async function loadOwnedAdviceRecord(adviceRecordId: number, adviserUserId: number) {
  const record = await db.query.adviceRecords.findFirst({
    where: and(
      eq(adviceRecords.id, adviceRecordId),
      eq(adviceRecords.adviserId, adviserUserId),
    ),
  });

  if (!record) {
    const err: any = new Error("ADVICE_RECORD_NOT_FOUND_OR_ACCESS_DENIED");
    err.status = 404;
    throw err;
  }

  return record;
}

function assertNotReviewPending(record: any) {
  if (record.status === "review_pending") {
    const err: any = new Error("RECORD_LOCKED_UNDER_REVIEW");
    err.status = 423;
    throw err;
  }
}

router.post(
  "/api/adviser/planner/:adviceRecordId/objectives",
  requireAuth,
  async (req: any, res, next) => {
    try {
      requireAdviser(req);

      const adviceRecordId = Number(req.params.adviceRecordId);
      const record = await loadOwnedAdviceRecord(adviceRecordId, req.user.id);
      assertNotReviewPending(record);

      const {
        objectiveType,
        description,
        targetAmount,
        targetDate,
        priority,
        supersedesObjectiveId,
      } = req.body;

      let insertedObjective: any;

      await writeWithVersioning(
        db,
        adviceRecordId,
        req.user.id,
        "client_objective_added",
        async (tx: any) => {
          const [created] = await tx.insert(clientObjectives).values({
            clientUserId: record.clientId,
            adviceRecordId,
            objectiveType,
            description,
            targetAmount,
            targetDate,
            priority: priority || "primary",
            createdBy: req.user.id,
          }).returning();

          insertedObjective = created;

          if (supersedesObjectiveId) {
            await tx.update(clientObjectives)
              .set({
                supersededAt: new Date(),
                supersededBy: created.id,
              })
              .where(and(
                eq(clientObjectives.id, supersedesObjectiveId),
                eq(clientObjectives.adviceRecordId, adviceRecordId),
              ));
          }
        }
      );

      res.status(201).json(insertedObjective);
    } catch (err) {
      next(err);
    }
  }
