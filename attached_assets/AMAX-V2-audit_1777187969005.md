### PATCH 5 — `server/routes.ts`: Add Google auth route

```typescript
// POST /api/auth/google
app.post("/api/auth/google", async (req, res) => {
  try {
    const { access_token } = req.body;
    if (!access_token) return res.status(400).json({ error: "Missing access_token." });

    // Verify token and fetch profile from Google
    const tokenInfo = await fetch(
      `https://oauth2.googleapis.com/tokeninfo?access_token=${access_token}`
    );
    const info = await tokenInfo.json() as any;
    if (!tokenInfo.ok || info.error) {
      return res.status(401).json({ error: "Invalid Google token." });
    }

    const expectedClientId = process.env.GOOGLE_CLIENT_ID;
    if (expectedClientId && info.aud !== expectedClientId) {
      return res.status(401).json({ error: "Token audience mismatch." });
    }

    const profileRes = await fetch(
      `https://www.googleapis.com/oauth2/v2/userinfo?access_token=${access_token}`
    );
    if (!profileRes.ok) throw new Error("Failed to fetch Google profile.");
    const profile = await profileRes.json() as any;
    const { email, given_name, family_name, id: googleSubId } = profile;

    if (!email) return res.status(400).json({ error: "Google account has no email." });

    let user = await storage.getUserByEmail(email.toLowerCase());
    let isNewUser = false;

    if (!user) {
      const base = email.split("@")[0].toLowerCase().replace(/[^a-z0-9_]/g, "_");
      let username = base;
      let attempt = 0;
      while (await storage.getUserByUsername(username)) { username = `${base}${++attempt}`; }

      user = await storage.createUser({
        username,
        email: email.toLowerCase(),
        password: null as any,
        firstName: given_name?.trim() || "AMAX",
        lastName:  family_name?.trim() || "User",
        googleId: googleSubId,
        emailVerified: true,
        kycStatus: "pending",
        userTier: "standard",
      });
      await storage.createWallet({ userId: user.id, currency: "AUD", balance: "0.00", availableBalance: "0.00", walletType: "fiat" });
      await writeAuditLog(user.id, "google_register", "user", String(user.id), { email }, req.ip || null);
      isNewUser = true;
    } else if (!(user as any).googleId) {
      await storage.updateUser(user.id, { googleId: googleSubId, emailVerified: true });
      await writeAuditLog(user.id, "google_link", "user", String(user.id), { email }, req.ip || null);
    }

    if (user.accountFrozen) return res.status(403).json({ error: "Account suspended. Contact support." });

    const token = signToken({ userId: user.id, username: user.email, email: user.email });
    res.json({ token, isNewUser, user: { id: user.id, email: user.email, firstName: user.firstName } });
  } catch (err: any) {
    console.error("[google-auth]", err.message);
    res.status(500).json({ error: "Google authentication failed." });
  }
});
```

export const smsConfigured = !!(process.env.NOTIFYRE_API_TOKEN);

export async function sendSmsOtp(phone: string, otp: string): Promise<{ sent: boolean }> {
  const token = process.env.NOTIFYRE_API_TOKEN;
  const from  = process.env.NOTIFYRE_FROM || "AMAXGLOBAL";

  if (!token) {
    console.log(`[sms-dev] OTP for ${phone}: ${otp}`);
    return { sent: false };
  }

  const to = phone.replace(/^\+/, "").replace(/\s/g, "");
  const res = await fetch("https://api.notifyre.com/sms/send", {
    method: "POST",
    headers: {
      "Authorization": `Bearer ${token}`,
      "Content-Type": "application/json",
      "x-notifyre-version": "2021-10-01",
    },
    body: JSON.stringify({
      Recipients: [{ to }],
      Body: `Your AMAX GLOBAL verification code is: ${otp}\nValid for 10 minutes. Do not share this code.`,
      From: from,
    }),
  });

  const data = await res.json() as any;
  if (!res.ok) {
    console.error("[sms] Notifyre error:", data);
    throw new Error("Failed to send SMS. Check the number and try again.");
  }
  return { sent: true };
}

// POST /api/auth/phone/send-otp
app.post("/api/auth/phone/send-otp", async (req, res) => {
  try {
    const { phone } = req.body;
    if (!phone) return res.status(400).json({ error: "Phone number required." });

    const normalised = "+" + phone.replace(/[\s\-()]/g, "").replace(/^\+/, "");
    if (!/^\+[1-9]\d{6,14}$/.test(normalised)) {
      return res.status(400).json({ error: "Enter a valid mobile number with country code e.g. +61412345678." });
    }

    const otp    = Math.floor(100000 + Math.random() * 900000).toString();
    const expiry = new Date(Date.now() + 10 * 60 * 1000);

    await db.execute(sql`
      CREATE TABLE IF NOT EXISTS phone_otps (
        phone      text PRIMARY KEY,
        otp        text NOT NULL,
        expires_at timestamp NOT NULL,
        created_at timestamp DEFAULT NOW()
      )
    `);
    await db.execute(sql`
      INSERT INTO phone_otps (phone, otp, expires_at)
      VALUES (${normalised}, ${otp}, ${expiry})
      ON CONFLICT (phone) DO UPDATE SET otp = ${otp}, expires_at = ${expiry}, created_at = NOW()
    `);

    const { sendSmsOtp, smsConfigured } = await import("./email");
    await sendSmsOtp(normalised, otp);

    res.json({ sent: smsConfigured, ...(!smsConfigured ? { devOtp: otp } : {}) });
  } catch (err: any) {
    console.error("[phone-send-otp]", err.message);
    res.status(500).json({ error: err.message ?? "Failed to send code." });
  }
});

// POST /api/auth/phone/verify-otp
app.post("/api/auth/phone/verify-otp", async (req, res) => {
  try {
    const { phone, code, firstName, lastName } = req.body;
    if (!phone || !code) return res.status(400).json({ error: "Phone and code are required." });

    const normalised = "+" + phone.replace(/[\s\-()]/g, "").replace(/^\+/, "");

    const rows = await db.execute(sql`
      SELECT otp, expires_at FROM phone_otps WHERE phone = ${normalised}
    `) as any;
    const row = rows?.rows?.[0] ?? rows?.[0];

    if (!row)                          return res.status(400).json({ error: "No code found. Request a new one." });
    if (new Date() > new Date(row.expires_at)) return res.status(400).json({ error: "Code expired. Request a new one." });
    if (code.trim() !== row.otp)       return res.status(400).json({ error: "Invalid code. Try again." });

    await db.execute(sql`DELETE FROM phone_otps WHERE phone = ${normalised}`);

    const [existing] = await db.select().from(users).where(eq(users.phoneNumber, normalised));
    let user = existing;
    let isNewUser = false;

    if (!user) {
      const suffix = normalised.replace(/\D/g, "").slice(-8);
      let username = `ph${suffix}`;
      let attempt = 0;
      while (await storage.getUserByUsername(username)) { username = `ph${suffix}_${++attempt}`; }

      user = await storage.createUser({
        username,
        email: `${normalised.replace(/\D/g, "")}@phone.amax`,
        password: null as any,
        firstName: firstName?.trim() || "AMAX",
        lastName:  lastName?.trim()  || "User",
        phoneNumber: normalised,
        emailVerified: true,
        kycStatus: "pending",
        userTier: "standard",
      });
      await storage.createWallet({ userId: user.id, currency: "AUD", balance: "0.00", availableBalance: "0.00", walletType: "fiat" });
      await writeAuditLog(user.id, "phone_register", "user", String(user.id), { phone: normalised }, req.ip || null);

  const [link] = await db.select().from(adviserClients).where(
    and(eq(adviserClients.adviserUserId, adviserUserId), eq(adviserClients.clientUserId, clientUserId))
  );

  try {
    const { userId } = requireAuth(req);
    const role = await getRole(userId);
    if (!["adviser", "admin", "compliance_admin"].includes(role)) {
      return res.status(403).json({ error: "Adviser access required." });

      : 0;

async function runAdviserTaskAutomation() {
  try {
    const allLinks = await db.select().from(adviserClients);
    for (const link of allLinks) {
      const [client] = await db.select({ id: users.id, kycStatus: users.kycStatus, createdAt: users.createdAt })
        .from(users).where(eq(users.id, link.clientUserId));
      if (!client) continue;

      // Trigger 1: KYC pending > 24h
      const ageHours = (Date.now() - new Date(client.createdAt!).getTime()) / 3_600_000;
      if (client.kycStatus !== "verified" && ageHours > 24) {
        const [exists] = await db.select({ id: adviserTasks.id }).from(adviserTasks).where(
          and(eq(adviserTasks.adviserUserId, link.adviserUserId),
              eq(adviserTasks.clientUserId, link.clientUserId),
              eq(adviserTasks.taskType, "kyc_pending"),
              eq(adviserTasks.status, "open"))
        );
        if (!exists) await db.insert(adviserTasks).values({
          adviserUserId: link.adviserUserId, clientUserId: link.clientUserId,
          taskType: "kyc_pending", title: "KYC pending — follow up with client",
          priority: "high", dueAt: new Date(Date.now() + 2 * 86_400_000),
        });
      }

      // Trigger 2: Fee consent expiring in 30 days
      const thirtyDays = new Date(Date.now() + 30 * 86_400_000);
      const expiring = await db.select().from(feeConsents).where(
        and(eq(feeConsents.adviserUserId, link.adviserUserId),
            eq(feeConsents.clientUserId, link.clientUserId),
            eq(feeConsents.status, "active"),
            sql`${feeConsents.expiresAt} <= ${thirtyDays}`)
      );
      for (const consent of expiring) {
        const [exists] = await db.select({ id: adviserTasks.id }).from(adviserTasks).where(
          and(eq(adviserTasks.adviserUserId, link.adviserUserId),
              eq(adviserTasks.clientUserId, link.clientUserId),
              eq(adviserTasks.taskType, "fee_consent_expiring"),
              eq(adviserTasks.status, "open"))
        );
        if (!exists) await db.insert(adviserTasks).values({
          adviserUserId: link.adviserUserId, clientUserId: link.clientUserId,
          taskType: "fee_consent_expiring",
          title: `Fee consent expiring ${new Date(consent.expiresAt).toLocaleDateString("en-AU")} — renewal required`,
          priority: "critical", dueAt: new Date(consent.expiresAt),
        });
      }

      // Trigger 3: Quarterly portfolio review
      const ninetyDaysAgo = new Date(Date.now() - 90 * 86_400_000);
      const [recentReview] = await db.select({ id: adviserTasks.id }).from(adviserTasks).where(
        and(eq(adviserTasks.adviserUserId, link.adviserUserId),
            eq(adviserTasks.clientUserId, link.clientUserId),
            eq(adviserTasks.taskType, "portfolio_review"),
            sql`${adviserTasks.createdAt} >= ${ninetyDaysAgo}`)
      );
      if (!recentReview) await db.insert(adviserTasks).values({
        adviserUserId: link.adviserUserId, clientUserId: link.clientUserId,
        taskType: "portfolio_review", title: "Quarterly portfolio review due",
        priority: "medium", dueAt: new Date(Date.now() + 7 * 86_400_000),
      });
    }
    console.log("[task-automation] completed at", new Date().toISOString());
  } catch (err: any) {
    console.error("[task-automation] failed:", err.message);
  }
}
setInterval(runAdviserTaskAutomation, 24 * 60 * 60 * 1000);
