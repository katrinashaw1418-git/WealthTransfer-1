Fee engine logic — mirroring Netwealth
Percentage-based fees are calculated daily and deducted monthly in arrears from client cash accounts. Financial Newswire Your fee engine should follow this exactly:
typescript// Daily fee accrual calculation
async function calculateDailyFeeAccrual(clientUserId: number, adviserUserId: number) {
  // 1. Get total AUM (sum of all active product holdings in AUD)
  const holdings = await db.select()
    .from(clientProductHoldings)
    .where(and(
      eq(clientProductHoldings.clientUserId, clientUserId),
      eq(clientProductHoldings.status, "active")
    ));
  const totalAUM = holdings.reduce((sum, h) => sum + parseFloat(h.currentValue ?? "0"), 0);

  // 2. Get applicable fee rule
  const [rule] = await db.select().from(adviserFeeRules)
    .where(and(
      eq(adviserFeeRules.adviserUserId, adviserUserId),
      eq(adviserFeeRules.clientUserId, clientUserId),
      eq(adviserFeeRules.isActive, true)
    ));
  if (!rule) return null;

  // 3. Calculate daily fee (annual rate / 365)
  let annualRate = parseFloat(rule.feePercent ?? "0");
  
  // Tiered rate logic
  if (rule.calculationBasis === "tiered_percentage" && rule.tierBreakpoints) {
    const tiers = rule.tierBreakpoints as Array<{ upTo: number; rate: number }>;
    annualRate = tiers.find(t => totalAUM <= t.upTo)?.rate 
      ?? tiers[tiers.length - 1].rate;
  }

  const dailyFee = (totalAUM * annualRate) / 365;
  const platformCut = dailyFee * parseFloat(rule.platformSplitPercent ?? "0.5");
  const adviserCut = dailyFee * parseFloat(rule.adviserSplitPercent ?? "0.5");

  return { totalAUM, dailyFee, platformCut, adviserCut, annualRate };
}
