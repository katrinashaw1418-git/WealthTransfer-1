export async function getOrCreateClientAccount(userId: number, currency: string) {
  const [existing] = await db
    .select()
    .from(accounts)
    .where(
      and(
        eq(accounts.userId, userId),
        eq(accounts.currency, currency),
        eq(accounts.accountType, "client")
      )
    )
    .limit(1);

  if (existing) return existing;

  const [created] = await db
    .insert(accounts)
    .values({
      userId,
      currency,
      accountType: "client",
      status: "active",
    })
    .returning();

  return created;
}

export async function getAccountBalance(accountId: number) {
  const [row] = await db
    .select({
      balance: sql<string>`
        COALESCE(SUM(
          CASE
            WHEN ${ledgerEntries.direction} = 'credit'
              THEN ${ledgerEntries.amount}
            ELSE -${ledgerEntries.amount}
          END
        ), 0)
      `,
    })
    .from(ledgerEntries)
    .where(eq(ledgerEntries.accountId, accountId));

  return row?.balance ?? "0";
}

export async function getUserCurrencyBalance(userId: number, currency: string) {
  const [row] = await db
    .select({
      balance: sql<string>`
        COALESCE(SUM(
          CASE
            WHEN ${ledgerEntries.direction} = 'credit'
              THEN ${ledgerEntries.amount}
            ELSE -${ledgerEntries.amount}
          END
        ), 0)
      `,
    })
    .from(ledgerEntries)
    .where(
      and(
        eq(ledgerEntries.userId, userId),
        eq(ledgerEntries.currency, currency)
      )
    );

  return row?.balance ?? "0";
}

export async function postLedgerEntries(
  transactionId: number,
  entries: LedgerEntryInput[]
) {
  if (entries.length < 2) {
    throw new Error("Ledger transaction must have at least two entries");
  }

  const currency = entries[0].currency;

  const totalCredits = entries
    .filter((e) => e.direction === "credit")
    .reduce((sum, e) => sum + Number(e.amount), 0);

  const totalDebits = entries
    .filter((e) => e.direction === "debit")
    .reduce((sum, e) => sum + Number(e.amount), 0);

  if (currency && entries.some((e) => e.currency !== currency)) {
    throw new Error("Multi-currency ledger entries require explicit FX transaction handling");
  }

  if (Math.abs(totalCredits - totalDebits) > 0.00000001) {
    throw new Error("Ledger entries are not balanced");
  }

  await db.insert(ledgerEntries).values(
    entries.map((entry) => ({
      transactionId,
      accountId: entry.accountId,
      userId: entry.userId,
      currency: entry.currency,
      direction: entry.direction,
      amount: entry.amount,
      description: entry.description,
    }))
  );
}

export function getIdempotencyKey(req: any): string {
  const key = req.headers["idempotency-key"];

  if (!key || typeof key !== "string") {
    throw new Error("Missing Idempotency-Key header");
  }

  if (key.length < 12) {
    throw new Error("Invalid Idempotency-Key");
  }

  return key;
}

export async function findTransactionByIdempotencyKey(key: string) {
  const [existing] = await db
    .select()
    .from(transactions)
    .where(eq(transactions.idempotencyKey, key))
    .limit(1);

  return existing ?? null;
}

type TransactionStatus =
  | "pending"
  | "processing"
  | "settled"
  | "failed"
  | "reversed";

const allowedTransitions: Record<TransactionStatus, TransactionStatus[]> = {
  pending: ["processing", "failed"],
  processing: ["settled", "failed"],
  settled: ["reversed"],
  failed: [],
  reversed: [],
};

export function assertValidTransition(
  current: TransactionStatus,
  next: TransactionStatus
) {
  if (!allowedTransitions[current]?.includes(next)) {
    throw new Error(`Invalid transaction transition: ${current} → ${next}`);
  }
}

export async function settleDepositTransaction(transactionId: number) {
  const [tx] = await db
    .select()
    .from(transactions)
    .where(eq(transactions.id, transactionId))
    .limit(1);

  if (!tx) throw new Error("Transaction not found");

  if (tx.status === "settled") {
    return tx;
  }

  if (tx.status !== "processing" && tx.status !== "pending") {
    throw new Error(`Cannot settle transaction in status ${tx.status}`);
  }

  const clientAccount = await getOrCreateClientAccount(tx.userId, tx.currency);
  const suspenseAccount = await getOrCreateSuspenseAccount(tx.currency);

  await postLedgerEntries(tx.id, [
    {
      accountId: clientAccount.id,
      userId: tx.userId,
      currency: tx.currency,
      direction: "credit",
      amount: tx.amount,
      description: "Client deposit settled",
    },
    {
      accountId: suspenseAccount.id,
      userId: suspenseAccount.userId,
      currency: tx.currency,
      direction: "debit",
      amount: tx.amount,
      description: "Partner confirmed receipt",
    },
  ]);

  const [updated] = await db
    .update(transactions)
    .set({
      status: "settled",
      settledAt: new Date(),
      updatedAt: new Date(),
    })
    .where(eq(transactions.id, tx.id))
    .returning();

  return updated;
}
